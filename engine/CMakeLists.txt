cmake_minimum_required(VERSION 3.18)
project(nr_student_cpu CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  # Sources contain UTF-8 comments and use std::vector/iostream exceptions.
  # Keep command-line builds identical to the configured Visual Studio build.
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/EHsc> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  set(CMAKE_CXX_FLAGS_RELEASE "/O2 /DNDEBUG")
else()
  set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
endif()

include_directories(include src)

find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)

# AVX2 数值内核(attn.cpp/gemm.cpp 开 AVX2,运行期 __builtin_cpu_supports 守卫回落
# 标量;其余 TU 保持基线 ISA,便于 MSVC/老机器移植)。
option(NR_AVX2 "Build the AVX2 microkernels (x86-64 hosts)" ON)
if(NR_AVX2 AND MSVC)
  set(NR_AVX2_FLAGS "/arch:AVX2")
elseif(NR_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
  set(NR_AVX2_FLAGS "-mavx2;-mfma")
else()
  set(NR_AVX2_FLAGS "")
endif()

# Compile each VNNI ISA separately; never raise the ISA of the dispatcher or
# image/engine code. Probe intrinsics at configure time (compile, never execute).
option(NR_VNNI "Build runtime-dispatched AVX-VNNI and AVX-512 VNNI kernels" ${NR_AVX2})
set(NR_VNNI_SOURCES "")
set(NR_BUILD_AVX_VNNI OFF)
set(NR_BUILD_AVX512_VNNI OFF)
if(NR_VNNI AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64|i.86")
  include(CheckCXXSourceCompiles)
  set(NR_SAVED_REQUIRED_FLAGS "${CMAKE_REQUIRED_FLAGS}")
  if(MSVC)
    set(NR_AVX_VNNI_FLAGS "/arch:AVX2")
    set(NR_AVX512_VNNI_FLAGS "/arch:AVX512")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
      list(APPEND NR_AVX_VNNI_FLAGS "/clang:-mavxvnni")
      list(APPEND NR_AVX512_VNNI_FLAGS "/clang:-mavx512vnni")
    endif()
  else()
    set(NR_AVX_VNNI_FLAGS "-mavx2;-mavxvnni")
    set(NR_AVX512_VNNI_FLAGS "-mavx512f;-mavx512dq;-mavx512cd;-mavx512bw;-mavx512vl;-mavx512vnni")
  endif()
  string(REPLACE ";" " " CMAKE_REQUIRED_FLAGS "${NR_AVX_VNNI_FLAGS}")
  set(CMAKE_REQUIRED_FLAGS "${NR_SAVED_REQUIRED_FLAGS} ${CMAKE_REQUIRED_FLAGS}")
  check_cxx_source_compiles("#include <immintrin.h>
    int main(int argc, char**) { __m256i a = _mm256_set1_epi32(argc);
      return _mm256_extract_epi32(_mm256_dpbusd_avx_epi32(a, a, a), 0); }" NR_CAN_BUILD_AVX_VNNI)
  string(REPLACE ";" " " CMAKE_REQUIRED_FLAGS "${NR_AVX512_VNNI_FLAGS}")
  set(CMAKE_REQUIRED_FLAGS "${NR_SAVED_REQUIRED_FLAGS} ${CMAKE_REQUIRED_FLAGS}")
  check_cxx_source_compiles("#include <immintrin.h>
    int main(int argc, char**) { __m512i a = _mm512_set1_epi32(argc);
      return _mm_cvtsi128_si32(_mm512_castsi512_si128(_mm512_dpbusd_epi32(a, a, a))); }" NR_CAN_BUILD_AVX512_VNNI)
  set(CMAKE_REQUIRED_FLAGS "${NR_SAVED_REQUIRED_FLAGS}")
  if(NR_CAN_BUILD_AVX_VNNI)
    set(NR_BUILD_AVX_VNNI ON)
    list(APPEND NR_VNNI_SOURCES src/gemm_vnni_avx.cpp)
    set_source_files_properties(src/gemm_vnni_avx.cpp PROPERTIES COMPILE_OPTIONS "${NR_AVX_VNNI_FLAGS}")
  endif()
  if(NR_CAN_BUILD_AVX512_VNNI)
    set(NR_BUILD_AVX512_VNNI ON)
    list(APPEND NR_VNNI_SOURCES src/gemm_vnni_avx512.cpp)
    set_source_files_properties(src/gemm_vnni_avx512.cpp PROPERTIES COMPILE_OPTIONS "${NR_AVX512_VNNI_FLAGS}")
  endif()
endif()
message(STATUS "VNNI kernels: AVX=${NR_BUILD_AVX_VNNI}, AVX512=${NR_BUILD_AVX512_VNNI}")

add_library(nr_cpu STATIC src/lanes.cpp src/engine.cpp src/engine_package.cpp src/engine_modern.cpp
                         src/attn.cpp src/gemm.cpp src/gemm_dispatch.cpp ${NR_VNNI_SOURCES})
if(NR_BUILD_AVX_VNNI)
  target_compile_definitions(nr_cpu PRIVATE NR_BUILD_AVX_VNNI=1)
endif()
if(NR_BUILD_AVX512_VNNI)
  target_compile_definitions(nr_cpu PRIVATE NR_BUILD_AVX512_VNNI=1)
endif()
if(MSVC)
  target_compile_options(nr_cpu PRIVATE /O2)
else()
  target_compile_options(nr_cpu PRIVATE -O3)
endif()
if(NR_AVX2_FLAGS)
  target_compile_definitions(nr_cpu PRIVATE NR_BUILD_AVX2=1)
  set_source_files_properties(src/attn.cpp src/gemm.cpp PROPERTIES COMPILE_OPTIONS "${NR_AVX2_FLAGS}")
endif()
# Field scalar tails must use the same separate multiply/add as their SIMD
# counterparts. Explicit FMA intrinsics in the GEMM kernels remain fused.
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  if(MSVC)
    set_property(SOURCE src/gemm.cpp APPEND PROPERTY COMPILE_OPTIONS "/clang:-ffp-contract=off")
  else()
    set_property(SOURCE src/gemm.cpp APPEND PROPERTY COMPILE_OPTIONS "-ffp-contract=off")
  endif()
endif()
target_link_libraries(nr_cpu OpenMP::OpenMP_CXX Threads::Threads)

add_executable(cpu_lanes tools/lanes_main.cpp)
target_link_libraries(cpu_lanes nr_cpu)

add_executable(cpu_engine tools/engine_main.cpp)
target_link_libraries(cpu_engine nr_cpu)
if(WIN32)
  target_link_libraries(cpu_engine windowscodecs ole32)
endif()

# Optional native CUDA backend; CPU-only builds still require no CUDA toolkit.
option(NR_CUDA "Build the NVIDIA CUDA backend and gpu_engine image CLI" OFF)
if(NR_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75 86)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(nr_cuda STATIC src/engine_cuda.cu)
  set_target_properties(nr_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(nr_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  if(MSVC)
    target_compile_options(nr_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
  endif()
  target_link_libraries(nr_cuda PUBLIC nr_cpu CUDA::cudart CUDA::cublas)
  target_compile_definitions(cpu_engine PRIVATE NR_HAS_CUDA=1)
  target_link_libraries(cpu_engine nr_cuda)
  add_executable(gpu_engine tools/engine_main.cpp)
  target_compile_definitions(gpu_engine PRIVATE NR_HAS_CUDA=1 NR_DEFAULT_CUDA=1)
  target_link_libraries(gpu_engine nr_cuda)
  add_executable(shared_moe_check tools/shared_moe_check.cu)
  set_target_properties(shared_moe_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(shared_moe_check PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  if(MSVC)
    target_compile_options(shared_moe_check PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
  endif()
  target_link_libraries(shared_moe_check CUDA::cudart)
  if(WIN32)
    target_link_libraries(gpu_engine windowscodecs ole32)
  endif()
endif()

add_executable(hash_probe tools/hash_probe.cpp)
target_link_libraries(hash_probe nr_cpu)

add_executable(gemm_bench tools/gemm_bench.cpp)
target_link_libraries(gemm_bench nr_cpu)

  target_link_libraries(kernel_check nr_cpu)
  add_test(NAME kernel_check COMMAND kernel_check)
  add_executable(vnni_check tools/vnni_check.cpp)
  target_link_libraries(vnni_check nr_cpu)
  add_test(NAME vnni_check COMMAND vnni_check)
